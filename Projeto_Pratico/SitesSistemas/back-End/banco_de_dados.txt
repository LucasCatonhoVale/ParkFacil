
CREATE DATABASE IF NOT EXISTS ParkFacil;

USE ParkFacil;


-- =========================================
-- TABELA: USUARIO
-- =========================================

CREATE TABLE Usuario (
    Id_Usuario INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
    NmUsuario VARCHAR(35) NOT NULL,
    SobreNome VARCHAR(100) NOT NULL,
    Cpf CHAR(11) NOT NULL UNIQUE,
    Telefone VARCHAR(15) NOT NULL,
    Email VARCHAR(150) NOT NULL UNIQUE,
    Senha VARCHAR(255) NOT NULL,
    Imagem mediumtext 
);


-- =========================================
-- TABELA: ESTACIONAMENTO
-- =========================================

CREATE TABLE Estacionamento (
    Id_Estacionamento integer NOT NULL AUTO_INCREMENT PRIMARY KEY,
    NmEstacionamento  VARCHAR(100) NOT NULL,
    Cnpj              CHAR(14) NOT NULL UNIQUE,
    QtVagas           int UNSIGNED NOT NULL,
    Telefone           VARCHAR(15) NOT NULL,
    Email               VARCHAR(150) NOT NULL UNIQUE,
    ValorHora           DECIMAL(10,2) NOT NULL,
    Descricao           TEXT,
    Endereco          VARCHAR(200) NOT NULL,
    Numero               VARCHAR(10) NOT NULL,
    Complemento       VARCHAR(100),
    Bairro               VARCHAR(100) NOT NULL,
    Cidade              VARCHAR(100) NOT NULL,
    Estado               CHAR(2) NOT NULL,
    Cep               CHAR(8) NOT NULL,
    HorarioAbertura   TIME NOT NULL,
    HorarioFechamento TIME NOT NULL,
);

-- =========================================
-- TABELA: CARRO
-- =========================================

CREATE TABLE Carro (
    Id_Carro   integer NOT NULL AUTO_INCREMENT PRIMARY KEY,
    Marca      VARCHAR(50) NOT NULL,
    Modelo     VARCHAR(100) NOT NULL,
    Placa      VARCHAR(7) NOT NULL UNIQUE,
    Cor        VARCHAR(30) NOT NULL,
    Tipo       VARCHAR(50) NOT NULL,
    Id_Usuario integer NOT NULL,
    CONSTRAINT FK_Carro_Usuario FOREIGN KEY (Id_Usuario)REFERENCES Usuario(Id_Usuario)
);


-- =========================================
-- TABELA: RESERVA
-- =========================================

CREATE TABLE Reserva (
    Id_Reserva           integer NOT NULL AUTO_INCREMENT PRIMARY KEY,
    DataHoraReserva      DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    DataHoraChegada      DATETIME NOT NULL,
    DataHoraSaida        DATETIME NOT NULL,
    StatusReserva        VARCHAR(30) NOT NULL DEFAULT 'PENDENTE',
    ValorTotal              DECIMAL(10,2) NOT NULL,
    Id_Carro              integer NOT NULL,
    Id_Estacionamento      integer NOT NULL,
    Id_Usuario             integer NOT NULL,
    CONSTRAINT FK_Reserva_Carro FOREIGN KEY (Id_Carro)REFERENCES Carro(Id_Carro),
    CONSTRAINT FK_Reserva_Estacionamento FOREIGN KEY (Id_Estacionamento)REFERENCES Estacionamento(Id_Estacionamento),
    CONSTRAINT FK_Reserva_Usuario FOREIGN KEY (Id_Usuario) REFERENCES Usuario(Id_Usuario)
);
-- =========================================
-- TABELA: IMAGEMESTACIONAMENTO
-- =========================================

CREATE TABLE ImagemEstacionamento (
    Id_Imagem 		 integer NOT NULL AUTO_INCREMENT PRIMARY KEY,
    CaminhoImagem    mediumtext NOT NULL,
    ImagemPrincipal  mediumtext NOT NULL ,
    Id_Estacionamento integer NOT NULL,
    CONSTRAINT FK_Imagem_Estacionamento FOREIGN KEY (Id_Estacionamento)REFERENCES Estacionamento(Id_Estacionamento)
);
